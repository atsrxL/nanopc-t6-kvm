cmake_minimum_required(VERSION 3.18)
project(t6_kvm VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(T6_BUILD_MPP "Build real hardware capture worker; requires pinned MPP" ON)
option(T6_SANITIZERS "Address/undefined sanitizers for offline C tests" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wformat=2 -Werror)
if(T6_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(t6-core STATIC native/pixels.c native/wire.c)
add_executable(t6-probe native/probe.c)
add_executable(t6-native-test native/test_native.c)
target_link_libraries(t6-native-test PRIVATE t6-core)
enable_testing()
add_test(NAME native-core COMMAND t6-native-test)
install(TARGETS t6-probe RUNTIME DESTINATION bin)
if(T6_BUILD_MPP)
  find_path(MPP_INCLUDE_DIR rk_mpi.h PATH_SUFFIXES rockchip REQUIRED)
  find_library(MPP_LIBRARY NAMES rockchip_mpp REQUIRED)
  add_executable(t6-capture native/capture.c)
  target_include_directories(t6-capture SYSTEM PRIVATE ${MPP_INCLUDE_DIR})
  target_link_libraries(t6-capture PRIVATE t6-core ${MPP_LIBRARY})
  set_target_properties(t6-capture PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  install(TARGETS t6-capture RUNTIME DESTINATION bin)
endif()
